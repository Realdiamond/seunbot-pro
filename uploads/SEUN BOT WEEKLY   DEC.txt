# ================= ROBUST REQUEST FUNCTION =================
def robust_request(url, headers=None, params=None, retries=5, backoff_factor=0.3, timeout=10):
    """Robust request with retries, user agent rotation, proxy rotation, and error handling"""
    session = requests.Session()
    retry = Retry(
        total=retries,
        read=retries,
        connect=retries,
        backoff_factor=backoff_factor,
        status_forcelist=(500, 502, 504, 429),
    )
    adapter = HTTPAdapter(max_retries=retry)
    session.mount('http://', adapter)
    session.mount('https://', adapter)
    
    if headers is None:
        headers = {}
    headers['User-Agent'] = random.choice(USER_AGENTS)
    
    # Proxy rotation
    if PROXIES:
        proxy = random.choice(PROXIES)
        session.proxies = {'http': proxy, 'https': proxy}
    else:
        session.proxies = {}  # No proxy if list empty
    
    try:
        response = session.get(url, headers=headers, params=params, timeout=timeout)
        response.raise_for_status()
        return response
    except requests.exceptions.ProxyError as errp:
        print(f"Proxy Error for {url}: {errp}")
    except requests.exceptions.HTTPError as errh:
        print(f"HTTP Error for {url}: {errh}")
    except requests.exceptions.ConnectionError as errc:
        print(f"Error Connecting for {url}: {errc}")
    except requests.exceptions.Timeout as errt:
        print(f"Timeout Error for {url}: {errt}")
    except requests.exceptions.RequestException as err:
        print(f"Request Exception for {url}: {err}")
    return None

# ================= SELENIUM SCRAPING FUNCTION =================
def get_page_with_selenium(url, wait_for_element=None, timeout=30):
    """Use Selenium for scraping when static request fails (handles JS)"""
    options = Options()
    options.add_argument('--headless')
    options.add_argument('--no-sandbox')
    options.add_argument('--disable-dev-shm-usage')
    options.add_argument(f'user-agent={random.choice(USER_AGENTS)}')
    
    driver = webdriver.Chrome(options=options)
    try:
        driver.get(url)
        if wait_for_element:
            WebDriverWait(driver, timeout).until(EC.presence_of_element_located((By.ID, wait_for_element)))
        html = driver.page_source
        return html
    except Exception as e:
        print(f"Selenium error for {url}: {e}")
        return None
    finally:
        driver.quit()

def get_binance_historical(symbol, interval, days):
    """Fetch historical OHLCV from Binance API with robustness"""
    url = "https://api.binance.com/api/v3/klines"
    limit = 1000
    end_time = int(time.time() * 1000)
    start_time = int((datetime.now() - timedelta(days=days)).timestamp() * 1000)
    data = []
    while start_time < end_time:
        params = {
            'symbol': symbol,
            'interval': interval,
            'startTime': start_time,
            'endTime': end_time,
            'limit': limit
        }
        response = robust_request(url, params=params)
        if response is None:
            return pd.DataFrame()
        new_data = json.loads(response.text)
        if not new_data:
            break
        data += new_data
        start_time = new_data[-1][0] + 1
    if not data:
        return pd.DataFrame()
    df = pd.DataFrame(data, columns=['Open time', 'Open', 'High', 'Low', 'Close', 'Volume', 'Close time', 'Quote asset volume', 'Number of trades', 'Taker buy base asset volume', 'Taker buy quote asset volume', 'Ignore'])
    df['Open time'] = pd.to_datetime(df['Open time'] / 1000, unit='s')
    df.set_index('Open time', inplace=True)
    df = df[['Open', 'High', 'Low', 'Close', 'Volume']]
    df = df.astype(float)
    return df

def get_historical_from_investing(symbol, start_date, end_date):
    """Sophisticated webscraping from investing.com for historical data (FX, stocks, NGX) with fallback parsing"""
    if '.NG' in symbol:
        pair_formatted = NGX_MAPPING.get(symbol, symbol.replace('.NG', '').lower()) + '-historical-data'
        base_url = "https://ng.investing.com/equities/"
    elif '=X' in symbol:
        pair_formatted = symbol.replace('=X', '').lower().replace('/', '-') + '-historical-data'
        base_url = "https://www.investing.com/currencies/"
    else:
        pair_formatted = symbol.lower() + '-historical-data'
        base_url = "https://www.investing.com/equities/"
    
    url = base_url + pair_formatted
    
    response = robust_request(url)
    if response is None:
        # Fallback to Selenium if request fails
        html = get_page_with_selenium(url, wait_for_element='curr_table')
        if html is None:
            return pd.DataFrame()
        soup = bs(html, 'html.parser')
    else:
        soup = bs(response.text, 'html.parser')
    
    table = soup.find('table', {'id': 'curr_table'})
    if not table:
        # Fallback: Try different ID or class
        table = soup.find('table', class_='historicalTbl')
        if not table:
            print(f"No table found for {symbol} on investing.com")
            return pd.DataFrame()
    
    rows = table.find('tbody').find_all('tr')
    data = []
    for row in rows:
        cols = row.find_all('td')
        if len(cols) >= 6:
            try:
                date_str = cols[0]['data-real-value']  # Unix timestamp
                date = datetime.fromtimestamp(int(date_str)).strftime('%Y-%m-%d')
                price = float(cols[1].text.replace(',', ''))
                open_ = float(cols[2].text.replace(',', ''))
                high = float(cols[3].text.replace(',', ''))
                low = float(cols[4].text.replace(',', ''))
                vol = cols[5].text
                data.append([date, open_, high, low, price, vol])
            except ValueError:
                continue
    
    if not data:
        return pd.DataFrame()
    
    df = pd.DataFrame(data, columns=['Date', 'Open', 'High', 'Low', 'Close', 'Volume'])
    df['Date'] = pd.to_datetime(df['Date'])
    df.set_index('Date', inplace=True)
    df.sort_index(inplace=True)
    
    # Filter by date range
    df = df[(df.index >= pd.to_datetime(start_date)) & (df.index <= pd.to_datetime(end_date))]
    
    return df

# ================= TECHNICAL INDICATORS =================
def calculate_ha_indicators(df):
    """Calculate technical indicators on Heikin-Ashi data including MACD"""
    df['ATR'] = talib.ATR(df['HA_High'], df['HA_Low'], df['HA_Close'], timeperiod=14)
    df['RSI'] = talib.RSI(df['HA_Close'], timeperiod=14)
    df['ADX'] = talib.ADX(df['HA_High'], df['HA_Low'], df['HA_Close'], timeperiod=14)
    df['MACD'], df['MACD_Signal'], df['MACD_Hist'] = talib.MACD(df['HA_Close'], fastperiod=12, slowperiod=26, signalperiod=9)
    df['Cycle_Phase'] = 'Bull' if df['HA_Close'].iloc[-1] > df['HA_Close'].iloc[-10] else 'Bear'
    df['Cycle_Duration'] = len(df[df['Cycle_Phase'] == df['Cycle_Phase'].iloc[-1]])
    df['Cycle_Momentum'] = (df['HA_Close'].iloc[-1] - df['HA_Close'].iloc[-10]) / df['HA_Close'].iloc[-10]
    return df

def detect_divergence(df, pivots, indicator='MACD'):
    """Detect MACD or volume divergence"""
    divergence = {'bullish': False, 'bearish': False}
    
    if len(pivots) < 2:
        return divergence
    
    recent_highs = [p for p in pivots[-4:] if p[2] == 'high']
    recent_lows = [p for p in pivots[-4:] if p[2] == 'low']
    
    if indicator == 'MACD':
        values = df['MACD'].values
    else:  # Volume
        values = df['Volume'].values
    
    # Bearish divergence: Price HH, indicator LH
    if len(recent_highs) >= 2:
        p1_idx, p1_price, _ = recent_highs[-2]
        p2_idx, p2_price, _ = recent_highs[-1]
        if p2_price > p1_price and values[p2_idx] < values[p1_idx]:
            divergence['bearish'] = True
    
    # Bullish divergence: Price LL, indicator HL
    if len(recent_lows) >= 2:
        p1_idx, p1_price, _ = recent_lows[-2]
        p2_idx, p2_price, _ = recent_lows[-1]
        if p2_price < p1_price and values[p2_idx] > values[p1_idx]:
            divergence['bullish'] = True
    
    return divergence

def detect_confluence(df, pivots):
    """Detect Smart Money Concepts confluence with divergences"""
    confluence = {'bullish_confluence': False, 'bearish_confluence': False, 'factors': []}
    
    # Order Blocks
    high_ob = df['HA_High'].rolling(window=5).max().shift(1)
    low_ob = df['HA_Low'].rolling(window=5).min().shift(1)
    if df['HA_Close'].iloc[-1] > high_ob.iloc[-1]: 
        confluence['factors'].append('Bullish OB')
    elif df['HA_Close'].iloc[-1] < low_ob.iloc[-1]: 
        confluence['factors'].append('Bearish OB')
    
    # Fair Value Gaps
    fvg_high = df['HA_High'].rolling(window=3).max().shift(2)
    fvg_low = df['HA_Low'].rolling(window=3).min().shift(2)
    if df['HA_Close'].iloc[-1] > fvg_high.iloc[-1] and df['HA_Close'].iloc[-2] < fvg_high.iloc[-1]: 
        confluence['factors'].append('Bullish FVG')
    elif df['HA_Close'].iloc[-1] < fvg_low.iloc[-1] and df['HA_Close'].iloc[-2] > fvg_low.iloc[-1]: 
        confluence['factors'].append('Bearish FVG')
    
    # Market Structure
    trend = 'Uptrend' if df['HA_Close'].iloc[-1] > df['HA_Close'].iloc[-10] else 'Downtrend'
    confluence['factors'].append(trend)
    
    # Add divergences
    macd_div = detect_divergence(df, pivots, 'MACD')
    vol_div = detect_divergence(df, pivots, 'Volume')
    
    if macd_div['bullish']:
        confluence['factors'].append('Bullish MACD Divergence')
    elif macd_div['bearish']:
        confluence['factors'].append('Bearish MACD Divergence')
    
    if vol_div['bullish']:
        confluence['factors'].append('Bullish Volume Divergence')
    elif vol_div['bearish']:
        confluence['factors'].append('Bearish Volume Divergence')
    
    confluence['bullish_confluence'] = any('Bullish' in f for f in confluence['factors'])
    confluence['bearish_confluence'] = any('Bearish' in f for f in confluence['factors'])
    return confluence
